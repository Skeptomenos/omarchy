import os

  print(f"REFUSED: {message}", file=sys.stderr)
  raise SystemExit(1)

def identity(metadata: os.stat_result) -> tuple[int, int, int, int, int, int, int, int, int]:
  return (
    metadata.st_dev,
    metadata.st_ino,
    metadata.st_uid,
    metadata.st_gid,

    metadata.st_nlink,
    metadata.st_size,
    metadata.st_mtime_ns,
