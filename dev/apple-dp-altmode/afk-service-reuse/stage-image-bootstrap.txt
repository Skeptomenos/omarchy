readonly wrapper_size=28236
readonly boot_root="/boot"
transaction=""

refuse() {
  printf 'REFUSED: %s\n' "$*" >&2
  exit 1
}

secure_root_directory() {
  local metadata owner group mode
  [[ -d $1 && ! -L $1 ]] || refuse "root directory is missing or a symlink: $1"
  metadata=$(/usr/bin/stat -c '%u %g %a' -- "$1") || refuse "cannot inspect root directory: $1"
  read -r owner group mode <<<"$metadata" || refuse "cannot parse root directory metadata"
  [[ $owner == "0" && $group == "0" && $mode =~ ^[0-7]{3,4}$ ]] ||
    refuse "root directory owner or mode changed: $1"
  (( (8#$mode & 0022) == 0 )) || refuse "root directory is group/world writable: $1"
}

(( EUID == 0 )) || refuse "root is required"
(( $# == 0 )) || refuse "no arguments are accepted"
[[ $PATH == "/usr/bin:/bin" && $LANG == "C.UTF-8" ]] ||
  refuse "bootstrap environment changed"
while IFS= read -r name; do
  case $name in
    LANG | PATH | PWD | SHLVL) ;;
    *) refuse "unexpected bootstrap environment: $name" ;;
  esac
done < <(compgen -e)
secure_root_directory "$boot_root"
transaction=$(/usr/bin/mktemp -d -- "$boot_root/.dev147-afk-reuse-stage.XXXXXXXXXX") ||

trap 'exit 143' TERM
